// --- Inbox Engine: Fully Activated --- // This Set keeps track of messages already on the screen so they don't duplicate or cause the screen to flash every 15 seconds let renderedMessageIds = new Set(); async function fetchStudentRequests() { try { const timeBuster = Date.now(); const response = await fetch('https://n8n.automania.cloud/webhook/reziflow-get-driver-msgs?t=' + timeBuster, { cache: 'no-store' }); // If the sheet is empty, n8n might return nothing or a 404. Handle gracefully. if (!response.ok) return; let rawData = await response.json(); // Handle if n8n returns a single object instead of an array const requests = Array.isArray(rawData) ? rawData : [rawData]; const feed = document.getElementById('message-feed'); // Remove the "Static Example" card if it's still there if (feed.innerHTML.includes('Static Example Message Card')) { feed.innerHTML = ''; } requests.forEach(req => { // Only render if we haven't seen this Message_ID yet and it has valid data if (req.Message_ID && !renderedMessageIds.has(req.Message_ID)) { // Select an icon based on the Type of request let iconHtml = ''; if (String(req.Type).includes('Stop')) iconHtml = ''; if (String(req.Type).includes('Delay')) iconHtml = ''; const cardHtml = `
${iconHtml}

${req.Type || 'Dispatch Note'}

From: ${req.Student_Name || 'Resident'}

${req.Time || '--:--'}

"${req.Message || ''}"

`; feed.insertAdjacentHTML('afterbegin', cardHtml); renderedMessageIds.add(req.Message_ID); } }); } catch (e) { console.log("Awaiting new dispatch messages..."); } } // Acknowledge and remove message from UI AND the Google Sheet async function clearMessage(btnElement, messageId) { btnElement.innerHTML = ''; btnElement.disabled = true; try { // Post to the n8n Resolution Engine to change status in the sheet await fetch('https://n8n.automania.cloud/webhook/reziflow-resolve-msg', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ID: messageId, Status: 'Acknowledged' }) }); } catch(e) { console.log("Failed to sync resolution to cloud."); } // Animate card shrinking and removing it from the DOM const card = btnElement.closest('.msg-card'); card.style.transition = "all 0.4s cubic-bezier(0.16, 1, 0.3, 1)"; card.style.opacity = "0"; card.style.transform = "translateY(-10px) scale(0.95)"; setTimeout(() => { card.remove(); // Optionally remove from Set if you want it to reappear if the server sends it again renderedMessageIds.delete(messageId); }, 400); } // Poll for new messages every 15 seconds setInterval(fetchStudentRequests, 15000); // Initial fetch on load fetchStudentRequests();
